    state TinjauRT <>

    MenungguRW --> TinjauRW: Ketua RW Meninjau Berkas
    state TinjauRW <>
    TinjauRW --> Ditolak: Ditolak RW (+ Alasan)
    TinjauRW --> MenungguDesa: Disetujui RW (Status: MENUNGGU_ADMIN_DESA)

    MenungguDesa --> TinjauDesa: Admin Desa Validasi Final
    state TinjauDesa <>
    TinjauDesa --> Ditolak: Ditolak Desa

    state Penerbitan {
        [*] --> GenerateNomor: Terbitkan Nomor Surat Resmi

        NotifWA --> [*]

    Penerbitan --> SiapUnduh: Surat Siap Diunduh Warga (PDF)

    participant DomPDF as 📄 DomPDF Rendering Engine

    participant WA as 📲 WhatsappNotificationService
    participant Fonnte as ☁️ Fonnte Cloud Gateway

    Admin->>Browser: Klik tombol 'Terbitkan Surat Resmi'
    Browser->>Router: POST /surat/{id}/approval (action=approve)
    Router->>Router: Verifikasi Middleware Auth & Spatie RBAC ('admin_desa')
    Router->>Controller: handleAction(request, surat)
    
    activate Controller
    Controller->>Controller: handleApproval(surat)
    Controller->>Model: Cek Status Aktif ('MENUNGGU_ADMIN_DESA')
    Controller->>Controller: Generate Nomor Surat: '470/{id}/RW.{rw}/{year}'
    
    Controller->>Model: update([status='SELESAI', approved_by_admin_desa_at=now()])
    activate Model
    Model->>DB: UPDATE pengajuan_surat SET status='SELESAI'...
    DB-->>Model: Query OK (Row Updated)
    deactivate Model

    %% PDF Generation
    Controller->>Controller: generatePdf(surat)
    Controller->>Controller: resolvePdfView(surat) -> 'surat.pdf.sktm'
    Controller->>DomPDF: loadView('surat.pdf.sktm', compact('surat'))
    activate DomPDF
    DomPDF->>DomPDF: Render HTML/CSS to Binary Stream A4
    DomPDF-->>Controller: Return Raw PDF Buffer
    deactivate DomPDF
    Controller->>Controller: Simpan PDF di storage/app/public/surat_xxx.pdf

    %% Dispatch Email
    Controller->>Controller: sendEmailSuratSelesai(surat)
    activate Mailer
    Controller->>Mailer: Mail::to(warga_email)->send(new SuratSelesaiMail(pdf))
    Mailer-->>Controller: Dispatched via SMTP (Port 587)
    deactivate Mailer

    %% Dispatch WhatsApp
    Controller->>WA: notifyStatusUpdate(surat, 'SELESAI')
    activate WA
    WA->>WA: formatPhoneNumber(warga->nomor_telepon)
    WA->>Fonnte: HTTP POST https://api.fonnte.com/send (Token, Target, Pesan)
    Fonnte-->>WA: Return JSON {status: true, id: 'msg_123'}
    WA-->>Controller: WhatsApp Dispatched
    deactivate WA

    Controller-->>Browser: Redirect /surat/{id} (Flash: 'Pengajuan surat berhasil diproses')
    deactivate Controller
    Browser-->>Admin: Render UI Updated (Badge SELESAI & Tombol 'Unduh PDF' Aktif)
